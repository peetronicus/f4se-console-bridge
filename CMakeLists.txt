cmake_minimum_required(VERSION 3.24)
project(F4SEConsoleBridge VERSION 0.1.0 LANGUAGES CXX)
set(F4SE_SOURCE_ROOT "" CACHE PATH "Unmodified official F4SE source checkout")
if(NOT EXISTS "${F4SE_SOURCE_ROOT}/f4se/PluginAPI.h")
  message(FATAL_ERROR "Set F4SE_SOURCE_ROOT to the official F4SE source checkout")
endif()
add_library(F4SEConsoleBridge SHARED src/bridge.cpp)
target_include_directories(F4SEConsoleBridge PRIVATE "${F4SE_SOURCE_ROOT}")
target_compile_features(F4SEConsoleBridge PRIVATE cxx_std_20)
target_compile_definitions(F4SEConsoleBridge PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(F4SEConsoleBridge PRIVATE /W4 /EHsc /utf-8)
set_property(TARGET F4SEConsoleBridge PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_link_libraries(F4SEConsoleBridge PRIVATE bcrypt shell32 ole32)
add_executable(bridge_protocol_test tests/protocol_test.cpp)
target_compile_features(bridge_protocol_test PRIVATE cxx_std_20)
target_include_directories(bridge_protocol_test PRIVATE src)
